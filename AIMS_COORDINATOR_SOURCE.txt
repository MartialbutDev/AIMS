export function CoordinatorPages({
  section,
  session,
  data,
  onLogout,
  onNavigate,
}: CoordinatorPagesProps) {
  // ========================================
  // DASHBOARD
  // ========================================

  switch (section) {
    case 'dashboard':
      return (
        <CoordinatorDashboardPage
          data={data}
          name={session.name}
          onNavigate={onNavigate}
        />
      )

    // ========================================
    // STUDENT MANAGEMENT
    // ========================================

    case 'students':
      return <StudentsPage />

    // ========================================
    // INTERNSHIP APPLICATIONS
    // ========================================

    case 'applications':
      return (
        <ApplicationsPage
          data={data}
        />
      )

    // ========================================
    // NOTIFICATIONS
    // ========================================

    case 'notifications':
      return <NotificationsPage />

    // ========================================
    // ACCOUNT SETTINGS
    // ========================================

    case 'settings':
      return (
        <AccountSettingsPage
          session={session}
          onLogout={onLogout}
        />
      )

    // ========================================
    // HELP CENTER
    // ========================================

    case 'help-center':
      return (
        <HelpCenterPage
          session={session}
        />
      )

    // ========================================
    // DEFAULT DASHBOARD
    // ========================================

    default:
      return (
        <CoordinatorDashboardPage
          data={data}
          name={session.name}
          onNavigate={onNavigate}
        />
      )

export function CoordinatorDashboardPage({

    (task) => !checkedTasks.includes(task)
  ).length

  const completedCount = data.tasks.length - pendingCount

  function toggleTask(task: string) {
    setCheckedTasks((current) =>
      current.includes(task)
        ? current.filter((item) => item !== task)
        : [...current, task]
    )
  }

  function metricDestination(metric: DashboardMetric) {
    const label = metric.label.toLowerCase()

function formatDate(value: string) {
  const date = new Date(value)
  return Number.isNaN(date.getTime())
    ? value
    : new Intl.DateTimeFormat(undefined, { dateStyle: 'medium' }).format(date)
}

function ProgressBar({ value, label }: { value: number; label: string }) {
  const percent = Math.max(0, Math.min(100, value))
  return (
    <div className="student-progress" aria-label={`${label}: ${percent}%`}>
      <div className="student-progress-track"><span style={{ width: `${percent}%` }} /></div>
      <strong>{percent}%</strong>
    </div>
  )
}

function EvaluationPanel({
  studentId,
  evaluation,
  onSaved,
}: {
  studentId: string
  evaluation: CoordinatorStudentDetail['evaluation']
  onSaved: (grade: number, notes: string) => void
}) {
  const [grade, setGrade] = useState(evaluation ? evaluation.grade.toFixed(2) : '')
  const [notes, setNotes] = useState(evaluation?.notes ?? '')
  const [isSaving, setIsSaving] = useState(false)
  const [error, setError] = useState('')
  const [saved, setSaved] = useState(false)

  async function submit(event: FormEvent<HTMLFormElement>) {
    event.preventDefault()
    setIsSaving(true)
    setError('')
    setSaved(false)
    try {
      const result = await saveCoordinatorEvaluation(studentId, Number(grade), notes)
      onSaved(result.grade, result.notes ?? '')
      setSaved(true)
    } catch (saveError) {
      setError(saveError instanceof Error ? saveError.message : 'Could not save the evaluation.')
    } finally {
      setIsSaving(false)
    }
  }

  return (
    <form className="panel coordinator-evaluation" onSubmit={submit}>
      <div className="panel-heading"><div><h2>Coordinator summary & grade</h2><p>Grade scale: 1.00 is highest; 5.00 is lowest.</p></div></div>
      <div className="evaluation-fields">
        <label>Final grade
          <input aria-label="Final grade" type="number" min="1" max="5" step="0.01" required value={grade} onChange={(event) => setGrade(event.target.value)} placeholder="e.g. 1.25" />
        </label>
        <label>Evaluation notes
          <textarea value={notes} onChange={(event) => setNotes(event.target.value)} maxLength={5000} rows={3} placeholder="Add a concise OJT performance summary…" />
        </label>
      </div>
      {error && <p className="form-error" role="alert">{error}</p>}
      {saved && <p className="evaluation-saved" role="status">Evaluation saved.</p>}
      <button className="primary-button" type="submit" disabled={isSaving}>{isSaving ? 'Saving…' : 'Save grade'} <Icon name="check" size={15} /></button>
    </form>
  )
}

export function StudentsPage() {
  const [students, setStudents] = useState<CoordinatorStudent[]>([])
  const [selectedId, setSelectedId] = useState('')
  const [detail, setDetail] = useState<CoordinatorStudentDetail | null>(null)
  const [search, setSearch] = useState('')
  const [page, setPage] = useState(1)
  const [total, setTotal] = useState(0)
  const [totalPages, setTotalPages] = useState(0)
  const [isLoadingList, setIsLoadingList] = useState(true)
  const [error, setError] = useState('')

  useEffect(() => {
    let current = true
    getCoordinatorStudents(page, search)
      .then((result) => {
        if (!current) return
        setStudents((existing) => page === 1 ? result.items : [...existing, ...result.items])
        setTotal(result.total)
        setTotalPages(result.total_pages)
        setSelectedId((currentId) => result.items.some((student) => student.id === currentId)
          ? currentId
          : result.items[0]?.id ?? '')
      })
      .catch((loadError: unknown) => {
        if (current) setError(loadError instanceof Error ? loadError.message : 'Could not load students.')
      })
      .finally(() => { if (current) setIsLoadingList(false) })
    return () => { current = false }
  }, [page, search])

  useEffect(() => {
    if (!selectedId) return
    let current = true
    getCoordinatorStudentDetail(selectedId)
      .then((studentDetail) => {
        if (!current) return
        setDetail(studentDetail)
      })
      .catch((loadError: unknown) => {
        if (!current) return
        setError(loadError instanceof Error ? loadError.message : 'Could not load student details.')
      })
    return () => { current = false }
  }, [selectedId])

  function searchStudents(value: string) {
    setStudents([])
    setSelectedId('')
    setDetail(null)
    setPage(1)
    setIsLoadingList(true)
    setError('')
    setSearch(value)
  }

  function loadMoreStudents() {
    setIsLoadingList(true)
    setError('')
    setPage((current) => current + 1)
  }

  function selectStudent(studentId: string) {
    setError('')
    setSelectedId(studentId)
  }

  function handleEvaluationSaved(grade: number, notes: string) {
    setDetail((current) => current ? {
      ...current,
      evaluation: { grade, notes, updated_at: new Date().toISOString() },
    } : current)
    setStudents((current) => current.map((student) => student.id === selectedId ? { ...student, grade } : student))
  }

  return (
    <section className="workspace-page coordinator-workspace">
      <div className="workspace-page-heading"><div><span className="welcome-eyebrow"><span className="eyebrow-dot" /> COORDINATOR WORKSPACE</span><h1>Students</h1><p>Review student documents, applications, OJT hours, DTR photos, and evaluations.</p></div><span className="coordinator-student-count">{total.toLocaleString()} students</span></div>
      {error && <p className="form-error user-import-error" role="alert">{error}</p>}
      <div className="coordinator-columns">

// The current FastAPI campaign endpoint accepts ONE target_user_id per request.
// This helper creates one campaign per user, with at most 4 concurrent requests.
type CampaignInput = Parameters<typeof createAdminNotificationCampaign>[0]
export type GroupCampaignInput = Omit<CampaignInput, 'target_user_id'>

export interface NotificationGroupSendResult {
  created: AdminNotificationCampaign[]
  failed: Array<{ user: AdminUser; reason: string }>
}

export async function createNotificationsForGroup(
  recipients: AdminUser[],
  input: GroupCampaignInput,
  onProgress?: (completed: number, total: number) => void,
): Promise<NotificationGroupSendResult> {
  if (recipients.length === 0) throw new Error('There are no active recipients in this group.')

  const created: AdminNotificationCampaign[] = []
  const failed: NotificationGroupSendResult['failed'] = []
  let completed = 0
  const uniqueRecipients = [...new Map(recipients.map((user) => [user.id, user])).values()]

  for (let index = 0; index < uniqueRecipients.length; index += 4) {
    const batch = uniqueRecipients.slice(index, index + 4)
    await Promise.all(batch.map(async (user) => {
      try {
        const campaign = await createAdminNotificationCampaign({
          ...input,
          target_user_id: user.id,
        })
        created.push(campaign)
      } catch (error) {
        failed.push({
          user,
          reason: error instanceof Error ? error.message : 'Request failed',
        })
      } finally {
        completed += 1
        onProgress?.(completed, uniqueRecipients.length)
      }
    }))
  }

  return { created, failed }
}

const API_BASE_URL = import.meta.env.VITE_API_BASE_URL ?? ''

function getAccessToken(): string | undefined {
  try {
    const session: unknown = JSON.parse(localStorage.getItem('aims-session') ?? 'null')
    if (
      typeof session === 'object' &&
      session !== null &&
      'accessToken' in session &&
      typeof session.accessToken === 'string'
    ) {
      return session.accessToken
    }
  } catch {
    return undefined
  }
  return undefined
}

export async function apiRequest<T>(
  path: string,
  options: RequestInit = {},
): Promise<T> {
  const headers = new Headers(options.headers)
  const isMultipart = typeof FormData !== 'undefined' && options.body instanceof FormData
  if (!headers.has('Content-Type') && !isMultipart) {
    headers.set('Content-Type', 'application/json')
  }
  const token = getAccessToken()
  if (token && !headers.has('Authorization')) headers.set('Authorization', `Bearer ${token}`)

  const response = await fetch(`${API_BASE_URL}${path}`, {
    ...options,
    headers,
  })

  if (!response.ok) {
    let message = `Request failed (${response.status}): ${response.statusText}`
    try {
      const errorBody: unknown = await response.json()
      if (
        typeof errorBody === 'object' &&
        errorBody !== null &&
        'detail' in errorBody
      ) {
        const detail = errorBody.detail
        message = Array.isArray(detail)
          ? detail.map((item) => typeof item === 'string' ? item : JSON.stringify(item)).join('; ')
          : typeof detail === 'string' ? detail : JSON.stringify(detail)
      }
    } catch {
      // Keep the HTTP status message when the response does not contain JSON.
    }
    throw new Error(message)
  }

  return response.json() as Promise<T>
}

export async function apiRequestBlob(path: string): Promise<Blob> {
  const headers = new Headers()
  const token = getAccessToken()
  if (token) headers.set('Authorization', `Bearer ${token}`)
  const response = await fetch(`${API_BASE_URL}${path}`, { headers })
  if (!response.ok) {
    throw new Error(`Could not load file (${response.status}): ${response.statusText}`)
  }
  return response.blob()
}

def _hours_summary(records: list[DTR]) -> dict:

        record
        for record in records
        if record.status == DTRStatus.APPROVED
    ]

        sum(

        2,

        0,

    remaining_hours = round(

    )

    for entry in approved:
        if entry.date is not None and (entry.total_hours or 0) > 0:
            day = entry.date.date()

            daily_hours[day] = (
                daily_hours.get(day, 0.0)
                + float(entry.total_hours)
            )

    average_daily_hours = (
        sum(daily_hours.values()) / len(daily_hours)
        if daily_hours
        else None
    )

    days_left = (
        math.ceil(remaining_hours / average_daily_hours)
        if remaining_hours and average_daily_hours
        else 0 if not remaining_hours else None
    )

    return {
        "required_hours": required_hours,
        "approved_hours": total_hours,
        "remaining_hours": remaining_hours,
        "progress_percent": progress,
        "remaining_percent": 100 - progress,
        "average_approved_hours_per_day": (
            round(average_daily_hours, 2)
            if average_daily_hours is not None
            else None
        ),
        "estimated_days_left": days_left,
        "approved_dtr_count": len(approved),
        "pending_dtr_count": sum(
            record.status in (
                DTRStatus.PENDING,
                DTRStatus.SUBMITTED,
            )
            for record in records
        ),
        "rejected_dtr_count": sum(
            record.status == DTRStatus.REJECTED
            for record in records
        ),
    }


def _file_response(
    relative_path: Optional[str],
    filename: Optional[str],
) -> FileResponse:
    if not relative_path or os.path.isabs(relative_path):
        raise HTTPException(
            status_code=404,
            detail="File is not available",
        )

    asset_root = os.path.realpath(settings.ASSET_DIR)
    file_path = os.path.realpath(
        os.path.join(asset_root, relative_path)
    )

    try:
        is_contained = (
            os.path.commonpath((asset_root, file_path))
            == asset_root
        )
    except ValueError:
        is_contained = False

    if not is_contained or not os.path.isfile(file_path):
        raise HTTPException(
            status_code=404,
            detail="File is not available",
        )

    return FileResponse(
        file_path,
        media_type=(
            mimetypes.guess_type(file_path)[0]
            or "application/octet-stream"
        ),
        filename=os.path.basename(filename or file_path),
        content_disposition_type="inline",
        headers={
            "Cache-Control": "private, no-store",
            "X-Content-Type-Options": "nosniff",
        },
    )

    students = (
        query.order_by(User.last_name, User.first_name)
        .offset((page - 1) * limit)

            "total_pages": math.ceil(total / limit),

                document,

                    else "missing"
                ),

            for kind in REQUIRED_DOCUMENT_TYPES

            item["status"] == "verified"

def _company_name_map(db: Session, items: List[Application]) -> dict[str, str]:
    ids = {a.company_id for a in items if a.company_id}

def _paginate(query, page: int, limit: int):

def _validate_document_ownership(
    db: Session,
    document_id: str | None,
    student_id: str,
    field_label: str,
) -> Document | None:
    if not document_id:
        return None
    doc = db.query(Document).filter(Document.id == document_id).first()
    if not doc:
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail=f"{field_label}: document not found",
        )
    if doc.uploaded_by != student_id:
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail=f"{field_label}: you can only attach your own documents",
        )
    if doc.verification_status != "verified":
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail=f"{field_label}: document must be verified before attaching",
        )
    return doc

@router.post("/", response_model=ApplicationResponse, status_code=status.HTTP_201_CREATED)
async def create_application(
    application_data: ApplicationCreate,
    db: Session = Depends(get_db),
    current_user: User = Depends(get_current_user),
):
    if current_user.role != UserRole.STUDENT:
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail="Only students can apply",
        )

    company = db.query(Company).filter(Company.id == application_data.company_id).first()
    if not company:
        raise HTTPException(status_code=404, detail="Company not found")

    existing = (
        db.query(Application)
        .filter(
            Application.student_id == current_user.id,
            Application.company_id == application_data.company_id,
            Application.position == application_data.position,
            Application.status.in_([
                ApplicationStatus.PENDING,
                ApplicationStatus.REVIEWING,
                ApplicationStatus.INTERVIEW,
            ]),
        )
        .first()
    )
    if existing:
        raise HTTPException(
            status_code=400,
            detail="You already have an active application for this position",
        )

    _validate_document_ownership(
        db, application_data.resume_id, current_user.id, "Resume"
    )
    _validate_document_ownership(
        db, application_data.application_letter_id, current_user.id, "Application letter"
    )

    application = Application(
        id=str(uuid.uuid4()),
        student_id=current_user.id,
        company_id=application_data.company_id,
        position=application_data.position,
        cover_letter=application_data.cover_letter,
        resume_id=application_data.resume_id,
        application_letter_id=application_data.application_letter_id,
        status=ApplicationStatus.PENDING,
    )
    db.add(application)
    db.commit()
    db.refresh(application)

    # ✅ Notify student that application was submitted
    _notify_application_status(db, application, ApplicationStatus.PENDING, company.name)

    return _to_response(application, company.name)

    db.refresh(application)

    company = db.query(Company).filter(Company.id == application.company_id).first()
    company_name = company.name if company else "the company"

    # ✅ Notify only if status actually changed
    if application.status != old_status:
        _notify_application_status(db, application, application.status, company_name)

        .first()
    )

        raise HTTPException(status_code=404, detail="File not found")

        ".pdf": "application/pdf",
        ".jpg": "image/jpeg",
        ".jpeg": "image/jpeg",
        ".png": "image/png",
        ".heic": "image/heic",
        ".bmp": "image/bmp",
        ".tiff": "image/tiff",
        ".webp": "image/webp",
        ".gif": "image/gif",
    }

    media_type = media_types.get(ext, "application/octet-stream")

    fallback_name = os.path.basename(file_path)
    display_name = document.original_filename or fallback_name

    safe_name = (
        display_name.replace('"', "")
        .replace("\n", "")
        .replace("\r", "")

    )

        safe_name = f"{os.path.splitext(safe_name)[0]}{ext}"

    print(f'📄 Serving file: {fallback_name} as "{safe_name}" ({media_type})')

    return FileResponse(
        file_path,
        media_type=media_type,
        headers={
            "Content-Disposition": f'inline; filename="{safe_name}"',
            "Accept-Ranges": "bytes",
            "Cache-Control": "public, max-age=3600",

@router.post("/", response_model=DTRResponse, status_code=status.HTTP_201_CREATED)
async def create_dtr(
    date: str = Form(...),
    time_in: str = Form(...),
    time_out: Optional[str] = Form(None),
    tasks_completed: Optional[str] = Form(None),
    notes: Optional[str] = Form(None),
    latitude: Optional[float] = Form(None),
    longitude: Optional[float] = Form(None),
    location_address: Optional[str] = Form(None),
    image_in: UploadFile = File(...),
    image_out: Optional[UploadFile] = File(None),
    db: Session = Depends(get_db),
    current_user: User = Depends(get_current_user),
):
    """Submit a new DTR entry with images (Time-In)."""
    if current_user.role != UserRole.STUDENT:
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail="Only students can submit DTR",
        )

    # Parse date/time strings
    try:
        date_obj = datetime.strptime(date, "%Y-%m-%d").date()
        time_in_obj = datetime.strptime(time_in, "%H:%M:%S").time()
        time_out_obj = (
            datetime.strptime(time_out, "%H:%M:%S").time() if time_out else None
        )
    except ValueError as e:
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail=f"Invalid date or time format: {str(e)}",
        )

    if time_out_obj and time_in_obj >= time_out_obj:
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail="Time in must be before time out",
        )

    # Prevent duplicate entries for the same day
    existing = (
        db.query(DTR)
        .filter(DTR.student_id == current_user.id, DTR.date == date_obj)
        .first()
    )
    if existing:
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail="DTR already submitted for this date",
        )

    # Save images
    upload_folder = os.path.join(settings.ASSET_DIR, "dtr_uploads")
    image_in_path = save_upload_file(image_in, upload_folder)
    image_out_path = save_upload_file(image_out, upload_folder) if image_out else None

    # Compute total hours if time_out given
    total_hours: Optional[float] = None
    dtr_status = DTRStatus.PENDING
    submitted_date = None
    if time_out_obj:
        dt_in = datetime.combine(date_obj, time_in_obj)
        dt_out = datetime.combine(date_obj, time_out_obj)
        total_seconds = (dt_out - dt_in).total_seconds()
        total_hours = round(total_seconds / 3600, 2)
        dtr_status = DTRStatus.SUBMITTED
        submitted_date = datetime.utcnow()

    dtr = DTR(
        id=str(uuid.uuid4()),
        student_id=current_user.id,
        date=date_obj,
        time_in=time_in_obj,
        time_out=time_out_obj,
        total_hours=total_hours,
        tasks_completed=tasks_completed,
        notes=notes,
        latitude=latitude,
        longitude=longitude,
        location_address=location_address,
        image_in_path=image_in_path,
        image_out_path=image_out_path,
        status=dtr_status,
        submitted_date=submitted_date,
    )

    db.add(dtr)
    db.commit()
    db.refresh(dtr)

    # ✅ Notify student that Time-In was recorded
    _fire_dtr_notification(db, dtr, dtr_status.value)

    return _to_response(dtr, _student_name(current_user))

@router.put("/{dtr_id}", response_model=DTRResponse)
async def update_dtr_status(
    dtr_id: str,
    update_data: DTRUpdate,
    db: Session = Depends(get_db),
    current_user: User = Depends(get_current_user),
):
    """Update DTR status (coordinator/admin only)."""
    if current_user.role not in [UserRole.COORDINATOR, UserRole.ADMIN]:
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail="Only coordinators and admins can update DTR status",
        )

    dtr = db.query(DTR).filter(DTR.id == dtr_id).first()
    if not dtr:
        raise HTTPException(status_code=404, detail="DTR entry not found")

    # Capture old status before applying the update
    old_status_val = dtr.status

    update_data_dict = update_data.model_dump(exclude_unset=True)

    if "status" in update_data_dict:
        if update_data_dict["status"] in [DTRStatus.APPROVED, DTRStatus.REJECTED]:
            dtr.approved_date = datetime.utcnow()

    for key, value in update_data_dict.items():
        setattr(dtr, key, value)

    dtr.updated_at = func.now()
    db.commit()
    db.refresh(dtr)

    new_status_val = dtr.status
    student = db.query(User).filter(User.id == dtr.student_id).first()

    # Notify only if status actually changed
    if old_status_val != new_status_val:
        _fire_dtr_notification(db, dtr, new_status_val.value)

    return _to_response(dtr, _student_name(student))

@router.put("/{journal_id}", response_model=JournalResponse)
async def update_journal(
    journal_id: str,
    journal_data: JournalUpdate,
    db: Session = Depends(get_db),
    current_user: User = Depends(get_current_user),
):
    journal = db.query(Journal).filter(Journal.id == journal_id).first()
    if not journal:
        raise HTTPException(404, "Journal not found")

    is_owner = current_user.id == journal.student_id
    is_staff = current_user.role in [UserRole.COORDINATOR, UserRole.ADMIN]

    if not is_owner and not is_staff:
        raise HTTPException(403, "Access denied")

    if is_owner and not is_staff and journal.status != JournalStatus.DRAFT:
        raise HTTPException(400, "Only draft journals can be edited")

    # ✅ Capture old status before applying the update
    old_status_val = journal.status

    updates = journal_data.model_dump(exclude_unset=True)

    if "status" in updates:
        new_status = updates["status"]
        if is_owner and not is_staff and new_status != JournalStatus.SUBMITTED:
            raise HTTPException(403, "Students can only submit journals")
        if new_status == JournalStatus.SUBMITTED:
            journal.submitted_date = datetime.utcnow()
        elif new_status in (JournalStatus.APPROVED, JournalStatus.REJECTED):
            journal.approved_date = datetime.utcnow()

    for key, value in updates.items():
        setattr(journal, key, value)

    journal.updated_at = func.now()
    db.commit()
    db.refresh(journal)

    new_status_val = journal.status
    student = db.query(User).filter(User.id == journal.student_id).first()

    # ✅ Notify only if status actually changed
    if old_status_val != new_status_val:
        _fire_journal_notification(db, journal, new_status_val.value)

    student_name = f"{student.first_name} {student.last_name}" if student else None
    return _to_response(journal, student_name)

    db: Session = Depends(get_db),
    current_user: User = Depends(require_role("admin")),
):
    recipient = db.query(User).filter(
        User.id == campaign_data.target_user_id,
        User.is_active.is_(True),
    ).first()
    if recipient is None:
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND,
            detail="Active target user not found.",
        )

    now = datetime.utcnow()
    deadline_at = (
        _to_utc_naive(campaign_data.deadline_at)
        if campaign_data.deadline_at is not None
        else None
    )
    if deadline_at is not None and deadline_at <= now:
        raise HTTPException(
            status_code=status.HTTP_422_UNPROCESSABLE_CONTENT,
            detail="The deadline must be in the future.",
        )

    scheduled_at = (
        _to_utc_naive(campaign_data.scheduled_at)
        if campaign_data.scheduled_at is not None
        else now
    )
    is_scheduled = campaign_data.scheduled_at is not None
    if is_scheduled and scheduled_at <= now:
        raise HTTPException(
            status_code=status.HTTP_422_UNPROCESSABLE_CONTENT,
            detail="Choose a future time to schedule this notification.",
        )

    campaign = NotificationCampaign(
        created_by=current_user.id,
        target_user_id=recipient.id,
        target_name=f"{recipient.first_name} {recipient.last_name}".strip(),
        target_email=recipient.email,
        type=campaign_data.type,
        title=campaign_data.title,
        message=campaign_data.message,
        deadline_at=deadline_at,
        scheduled_at=scheduled_at,
        status="scheduled",
    )
    db.add(campaign)
    db.flush()
    if not is_scheduled:
        deliver_campaign(db, campaign, now)
    db.commit()
    db.refresh(campaign)
    return campaign

"""Authentication helpers for AIMS."""

from datetime import datetime, timedelta, timezone
from typing import Any, Dict, Optional

from fastapi import Depends, HTTPException, Query, status
from fastapi.security import OAuth2PasswordBearer
from jose import JWTError, jwt
from passlib.context import CryptContext
from sqlalchemy.orm import Session

from app.core.config import settings
from app.core.database import get_db
from app.models.user import User, UserRole


pwd_context = CryptContext(
    schemes=["argon2", "bcrypt"],
    deprecated="auto"
)

oauth2_scheme = OAuth2PasswordBearer(
    tokenUrl="/api/v1/auth/login/form"
)

optional_oauth2_scheme = OAuth2PasswordBearer(
    tokenUrl="/api/v1/auth/login/form",
    auto_error=False
)


def verify_password(
    plain_password: str,
    hashed_password: str
) -> bool:
    return pwd_context.verify(
        plain_password,
        hashed_password
    )


def get_password_hash(password: str) -> str:
    return pwd_context.hash(password)


def create_access_token(
    data: Dict[str, Any],
    expires_delta: Optional[timedelta] = None
) -> str:
    claims = data.copy()

    claims["exp"] = datetime.now(timezone.utc) + (
        expires_delta
        if expires_delta is not None
        else timedelta(
            minutes=settings.ACCESS_TOKEN_EXPIRE_MINUTES
        )
    )

    return jwt.encode(
        claims,
        settings.SECRET_KEY,
        algorithm=settings.ALGORITHM
    )


def decode_token(token: str) -> Dict[str, Any]:
    try:
        return jwt.decode(
            token,
            settings.SECRET_KEY,
            algorithms=[settings.ALGORITHM]
        )

    except JWTError as exc:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Invalid or expired token",
            headers={"WWW-Authenticate": "Bearer"}
        ) from exc


def _user_from_access_token(
    token: str,
    db: Session
) -> User:
    payload = decode_token(token)

    # Password-reset and verification tokens cannot access APIs.
    if payload.get("purpose") not in (None, "access"):
        raise HTTPException(
            status_code=401,
            detail="Invalid access token"
        )

    user_id = payload.get("sub")

    if not isinstance(user_id, str) or not user_id:
        raise HTTPException(
            status_code=401,
            detail="Invalid access token"
        )

    user = db.query(User).filter(
        User.id == user_id
    ).first()

    if user is None or not user.is_active:
        raise HTTPException(
            status_code=401,
            detail="Invalid authentication credentials"
        )

    return user


async def get_current_user(
    token: str = Depends(oauth2_scheme),
    db: Session = Depends(get_db)
) -> User:
    return _user_from_access_token(token, db)


async def get_current_user_from_token(
    token: Optional[str] = Query(default=None),
    header_token: Optional[str] = Depends(
        optional_oauth2_scheme
    ),
    db: Session = Depends(get_db)
) -> User:
    """
    Legacy image endpoints.

    Prefer Authorization headers.
    Query-string tokens should be phased out because
    they can leak through URLs and server logs.
    """

    chosen = header_token or token

    if not chosen:
        raise HTTPException(
            status_code=401,
            detail="Not authenticated"
        )

    return _user_from_access_token(chosen, db)


def require_role(required_role: str):
    """Require a role, with an administrator override."""

    async def role_checker(
        current_user: User = Depends(get_current_user)
    ) -> User:
        role = UserRole(current_user.role)

        if role not in (
            UserRole(required_role),
            UserRole.ADMIN
        ):
            raise HTTPException(
                status_code=403,
                detail="Insufficient permissions"
            )

        return current_user

    return role_checker
